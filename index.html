/* 
 * Slynks Hydroponic Controller - ESP32 Firmware v2.4
 * Auto-generated for Slynks Cloud & Blynk IoT
 */
#define BLYNK_TEMPLATE_ID "TMPL_SLYNKS_HYDRO"
#define BLYNK_DEVICE_NAME "Slynks Hydroponic Controller"
#define BLYNK_AUTH_TOKEN "slynks_hydro_pk_auth_9918274a"

#include <WiFi.h>
#include <WiFiClient.h>
#include <BlynkSimpleEsp32.h>
#include <OneWire.h>
#include <DallasTemperature.h>

char auth[] = BLYNK_AUTH_TOKEN;
char ssid[] = "Your_WiFi_SSID";
char pass[] = "Your_WiFi_Password";

// Pin Allocations
#define PH_PIN 34
#define EC_PIN 35
#define WATER_TEMP_PIN 4
#define PUMP_RELAY_PIN 25
#define LED_PWM_PIN 18
#define DOSER_PH_DOWN_PIN 26
#define DOSER_NUT_A_PIN 27

BlynkTimer timer;

void sendHydroTelemetry() {
  float rawPh = analogRead(PH_PIN);
  float ph = (rawPh * 3.3 / 4095.0) * 3.5; // Calibrated
  float ec = analogRead(EC_PIN) * 0.0008;
  
  Blynk.virtualWrite(V0, ph);
  Blynk.virtualWrite(V3, ec);
}

BLYNK_WRITE(V1) {
  int relayState = param.asInt();
  digitalWrite(PUMP_RELAY_PIN, relayState);
}

BLYNK_WRITE(V6) { // pH Down Doser Pulse
  int ml = param.asInt();
  if (ml > 0) {
    digitalWrite(DOSER_PH_DOWN_PIN, HIGH);
    delay(ml * 1000); // 1 sec per ml
    digitalWrite(DOSER_PH_DOWN_PIN, LOW);
  }
}

void setup() {
  Serial.begin(115200);
  pinMode(PUMP_RELAY_PIN, OUTPUT);
  pinMode(DOSER_PH_DOWN_PIN, OUTPUT);
  Blynk.begin(auth, ssid, pass);
  timer.setInterval(1000L, sendHydroTelemetry);
}

void loop() {
  Blynk.run();
  timer.run();
}
