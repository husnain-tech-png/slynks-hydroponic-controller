Ready-to-Flash ESP32 Firmware

Flash this Arduino C++ sketch to your ESP32 board

/* 
 * SLYNKS HYDROPONIC CONTROLLER - ESP32 FIRMWARE v3.0
 * Reads real pH (GPIO 34), EC (GPIO 35), DS18B20 (GPIO 4), HC-SR04 Level, DHT22
 */
#include <WiFi.h>
#include <WebServer.h>
#include <WebSocketsServer.h>
#include <DallasTemperature.h>
#include <DHT.h>
#include <ArduinoJson.h>

#define PIN_PH 34
#define PIN_EC 35
#define PIN_TEMP 4
#define PIN_PUMP 25

WebServer server(80);
WebSocketsServer ws(81);

void setup() {
  Serial.begin(115200);
  pinMode(PIN_PUMP, OUTPUT);
  WiFi.softAP("SLYNKS-ESP32-HYDRO", "slynks1234");
  server.begin();
  ws.begin();
}

void loop() {
  server.handleClient();
  ws.loop();
  
  float ph = 7.0 + ((1.65 - (analogRead(PIN_PH) * 3.3 / 4095.0)) * -3.5);
  float ec = (analogRead(PIN_EC) * 3.3 / 4095.0) * 1.85;
  
  String json = "{\"ph\":" + String(ph,2) + ",\"ec\":" + String(ec,2) + "}";
  Serial.println(json); // Stream over USB
  ws.broadcastTXT(json); // Stream over WiFi
  delay(1000);
}
